<script>

// ================== ВАШ URL ==================

const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbx7SGMITBwfW6KGj_aP84NgdBW-Rswy59aoLfdKYLyL5-5CRJG2QEgy3rRUtc3Zv0aVQA/exec';

// =============================================


const STATUSES = ['todo', 'work', 'done'];

const PRIORITY_ORDER = { high: 0, medium: 1, low: 2 };

let tasks = [];

let editingTask = null;


const form = document.getElementById('form');

const input = document.getElementById('input');

const colSelect = document.getElementById('col-select');

const overlay = document.getElementById('modal-overlay');

const modalClose = document.getElementById('modal-close');

const mTitleDisplay = document.getElementById('m-title-display');

const mText = document.getElementById('m-text');

const mDesc = document.getElementById('m-desc');

const mTag = document.getElementById('m-tag');

const mPriority = document.getElementById('m-priority');

const mDeadline = document.getElementById('m-deadline');

const mSteps = document.getElementById('m-steps');

const mNewStep = document.getElementById('m-new-step');

const mAddStep = document.getElementById('m-add-step');

const mSave = document.getElementById('m-save');


function send(payload) {

  return fetch(SCRIPT_URL, {

    method: 'POST',

    mode: 'no-cors',

    headers: { 'Content-Type': 'text/plain;charset=utf-8' },

    body: JSON.stringify(payload),

  }).catch(err => {

    console.error('Ошибка синхронизации:', err);

    alert('Не удалось сохранить изменения. Проверь интернет.');

  });

}


async function apiGet() {

  const res = await fetch(SCRIPT_URL + '?_=' + Date.now());

  if (!res.ok) throw new Error('GET failed: ' + res.status);

  const data = await res.json();

  return Array.isArray(data) ? data : [];

}


// важные выше, потом ближайшие дедлайны

function sortTasks(list) {

  return list.slice().sort((a, b) => {

    const pa = PRIORITY_ORDER[a.priority] !== undefined ? PRIORITY_ORDER[a.priority] : 1;

    const pb = PRIORITY_ORDER[b.priority] !== undefined ? PRIORITY_ORDER[b.priority] : 1;

    if (pa !== pb) return pa - pb;

    const da = a.deadline || '9999-12-31';

    const db = b.deadline || '9999-12-31';

    if (da !== db) return da < db ? -1 : 1;

    return String(a.id) < String(b.id) ? -1 : 1;

  });

}


function updateTaskStatus(task, newStatus) {

  task.status = newStatus;

  render();

  send({ action: 'update', id: task.id, fields: { status: newStatus } });

}


function getProgress(steps) {

  if (!steps || !steps.length) return null;

  return Math.round((steps.filter(s => s.done).length / steps.length) * 100);

}

function formatDate(d) {

  if (!d) return '';

  const parts = d.split('-');

  return parts[2] + '.' + parts[1];

}


function renderCard(t) {

  const card = document.createElement('div');

  card.className = 'card';

  card.draggable = true;


  const header = document.createElement('div');

  header.className = 'card-header';


  const dot = document.createElement('div');

  dot.className = 'priority-dot priority-' + (t.priority || 'medium');


  const title = document.createElement('div');

  title.className = 'card-title';

  title.textContent = t.text;


  header.append(dot, title);

  if (t.tag) {

    const badge = document.createElement('span');

    badge.className = 'tag-badge';

    badge.textContent = t.tag;

    header.appendChild(badge);

  }

  card.appendChild(header);


  const progress = getProgress(t.steps);

  if (progress !== null) {

    const bar = document.createElement('div');

    bar.className = 'progress-bar';

    const fill = document.createElement('div');

    fill.className = 'progress-fill';

    fill.style.width = progress + '%';

    bar.appendChild(fill);

    card.appendChild(bar);

  }


  const meta = document.createElement('div');

  meta.className = 'card-meta';

  if (t.deadline) meta.textContent = '📅 ' + formatDate(t.deadline);

  if (progress !== null) meta.textContent += (meta.textContent ? '  ·  ' : '') + progress + '% выполнено';

  if (meta.textContent) card.appendChild(meta);


  const idx = STATUSES.indexOf(t.status);

  const actions = document.createElement('div');

  actions.className = 'card-actions';

  if (idx > 0) {

    const b = document.createElement('button');

    b.textContent = '←';

    b.onclick = (e) => { e.stopPropagation(); updateTaskStatus(t, STATUSES[idx - 1]); };

    actions.appendChild(b);

  }

  if (idx < STATUSES.length - 1) {

    const b = document.createElement('button');

    b.textContent = '→';

    b.onclick = (e) => { e.stopPropagation(); updateTaskStatus(t, STATUSES[idx + 1]); };

    actions.appendChild(b);

  }

  const del = document.createElement('button');

  del.textContent = '🗑';

  del.onclick = (e) => {

    e.stopPropagation();

    if (!confirm('Удалить задачу?')) return;

    tasks = tasks.filter(x => x.id !== t.id);

    render();

    send({ action: 'delete', id: t.id });

  };

  actions.appendChild(del);

  card.appendChild(actions);


  card.addEventListener('click', () => openModal(t));

  card.addEventListener('dragstart', e => e.dataTransfer.setData('text/plain', t.id));

  return card;

}


function render() {

  document.querySelectorAll('.column').forEach(col => {

    const status = col.dataset.status;

    const box = col.querySelector('.cards');

    const list = sortTasks(tasks.filter(t => t.status === status));

    col.querySelector('.count').textContent = list.length;

    box.innerHTML = '';

    if (!list.length) { box.innerHTML = '<p class="empty">пусто</p>'; return; }

    list.forEach(t => box.appendChild(renderCard(t)));

  });

}


document.querySelectorAll('.column').forEach(col => {

  col.addEventListener('dragover', e => { e.preventDefault(); col.classList.add('over'); });

  col.addEventListener('dragleave', () => col.classList.remove('over'));

  col.addEventListener('drop', e => {

    e.preventDefault();

    col.classList.remove('over');

    const id = e.dataTransfer.getData('text/plain');

    const task = tasks.find(x => String(x.id) === id);

    if (task && task.status !== col.dataset.status) {

      updateTaskStatus(task, col.dataset.status);

    }

  });

});


form.onsubmit = (e) => {

  e.preventDefault();

  const text = input.value.trim();

  if (!text) return;

  tasks.push({

    id: 'tmp' + Date.now(),

    text, status: colSelect.value,

    description: '', tag: '', priority: 'medium', deadline: '', steps: [],

  });

  input.value = '';

  render();

  send({ action: 'add', task: { text, status: colSelect.value } });

  setTimeout(() => apiGet().then(data => { tasks = data; render(); }).catch(() => {}), 1500);

};


function renderSteps(steps) {

  mSteps.innerHTML = '';

  (steps || []).forEach((s, i) => {

    const li = document.createElement('li');

    li.className = 'step-item' + (s.done ? ' done' : '');

    const cb = document.createElement('input');

    cb.type = 'checkbox';

    cb.checked = s.done;

    cb.onchange = () => { s.done = cb.checked; li.classList.toggle('done', s.done); };

    const span = document.createElement('span');

    span.textContent = s.text;

    const rm = document.createElement('button');

    rm.className = 'remove-step';

    rm.textContent = '✕';

    rm.onclick = () => { steps.splice(i, 1); renderSteps(steps); };

    li.append(cb, span, rm);

    mSteps.appendChild(li);

  });

}


function openModal(task) {

  editingTask = task;

  mTitleDisplay.textContent = task.text;

  mText.value = task.text || '';

  mDesc.value = task.description || '';

  mTag.value = task.tag || '';

  mPriority.value = ['high', 'medium', 'low'].includes(task.priority) ? task.priority : 'medium';

  mDeadline.value = task.deadline || '';

  editingTask._stepsCopy = JSON.parse(JSON.stringify(task.steps || []));

  renderSteps(editingTask._stepsCopy);

  overlay.classList.add('open');

}

function closeModal() { overlay.classList.remove('open'); editingTask = null; }

modalClose.onclick = closeModal;

overlay.onclick = e => { if (e.target === overlay) closeModal(); };

document.addEventListener('keydown', e => { if (e.key === 'Escape') closeModal(); });


mAddStep.onclick = () => {

  const text = mNewStep.value.trim();

  if (!text || !editingTask) return;

  editingTask._stepsCopy.push({ text, done: false });

  mNewStep.value = '';

  renderSteps(editingTask._stepsCopy);

};

mNewStep.addEventListener('keydown', e => { if (e.key === 'Enter') { e.preventDefault(); mAddStep.click(); } });


mSave.onclick = () => {

  if (!editingTask) return;

  const task = editingTask;

  const updates = {

    text: mText.value.trim(),

    description: mDesc.value.trim(),

    tag: mTag.value.trim(),

    priority: mPriority.value,

    deadline: mDeadline.value,

    steps: task._stepsCopy,

  };

  if (!updates.text) return;

  Object.assign(task, updates);

  closeModal();

  render();

  send({ action: 'update', id: task.id, fields: updates });

};

if ('serviceWorker' in navigator) {

  window.addEventListener('load', () => {

  });

}


apiGet()

  .then(data => { tasks = data; render(); console.log('Загружено задач:', tasks.length); })

  .catch(err => {

    console.error(err);

    alert('Не удалось загрузить задачи: ' + err.message);

  });

</script>
